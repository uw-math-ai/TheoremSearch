% Matching pipeline: bidirectional NL <-> formal-Lean retrieval on one example
% pair (PFR blueprint Lemma 2.17 / ProbabilityTheory.measureMutualInfo_nonneg).
%
% Layout (Version B: strict horizontal, two stacked rows):
%
%   Row 1 (informal):  paper → LaTeX statement → slogan → embedding ─┐
%                                                                      cosine
%   Row 2 (formal):   Lean proj → Lean source → slogan → embedding   ─┘
%
%   Both rows read left→right, with purely horizontal arrows.
%   Both rows converge via short angled arrows into a cosine node at the right.
%   Model tags sit above arrows (informal) and below arrows (formal) to avoid
%   overlap. Embedding vectors shown as bmatrix column vectors.
%
% Build:  pdflatex -interaction=nonstopmode -output-directory=../out matching_pipeline_horizontal.tex
% Output: ../out/matching_pipeline_horizontal.pdf

\documentclass[tikz,border=8pt]{standalone}
\usepackage{tikz}

% Helper: tall column-vector display
\newcommand{\embcol}[6]{%
  $\begin{bmatrix} #1 \\ #2 \\ #3 \\ \vdots \\ #4 \\ #5 \\ #6 \end{bmatrix}$%
}

% --- Palette (figure_style.md section 3) ---------------------------------
\definecolor{BluePrimary}{HTML}{2E5C8A}

% --- Styles --------------------------------------------------------------
\tikzset{
  every node/.style       = {font=\sffamily\small, text=GrayText},
  procbox_informal/.style = {rectangle, rounded corners=3pt,
                             draw=AccentRed!70, dashed, line width=0.6pt,
                             fill=AccentRed!5,
                             inner sep=4pt, align=left},
  procbox_formal/.style   = {rectangle, rounded corners=3pt,
                             draw=BluePrimary!70, dashed, line width=0.6pt,
                             fill=BlueSoft!18,
                             inner sep=4pt, align=left},
  embvec/.style           = {inner sep=2pt, align=center,
                             font=\ttfamily\footnotesize,
                             text=GrayText},
  cosnode/.style          = {circle,
                             draw=AccentGreen, line width=1pt,
                             fill=AccentGreen!25,
                             minimum size=28mm,
                             font=\sffamily\small,
                             align=center, inner sep=1pt},
  arr/.style              = {-{Latex[length=2mm]}, line width=0.8pt,
                             draw=GrayText},
  arrconv/.style          = {-{Latex[length=2mm]}, line width=0.8pt,
                             draw=GrayText},
  modeltag/.style         = {font=\sffamily\footnotesize,
                             text=GrayText, inner sep=1pt,
                             align=center},
  armlabel/.style         = {font=\sffamily\bfseries\small\itshape,
                             inner sep=1pt},
  logo/.style             = {inner sep=0pt, outer sep=0pt},
}
